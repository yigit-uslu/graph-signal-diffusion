% ugnn_fig2_wide.tex -- Fig 2 (WIDE variant): one encoder block + one decoder block of the
% U-GNN, oriented as Fig 1's U.  Both blocks are HORIZONTAL chains (straight internal arrows).
%   Encoder (left):  V_b enters TOP-LEFT, chain Pi -> Phi -> split (Z_b out, bottom-right into
%     the down-sampler).  The learned selector Psi_b -> TopK is STACKED under the split (Z_b
%     drops straight in); it emits C_{b+1}.
%   Decoder (right): chain Pi^skip -> Pi -> Phi (flat).  C_{b+1}^T Y_{b+1} rises into Pi^skip
%     from below; the skip Z_b feeds straight in from the LEFT (direct from the encoder Z_b);
%     Y_b exits TOP-RIGHT.
%   Junction (centre): a small U -- C_{b+1}Z_b points DOWN, the elided deep levels (dotted U +
%     annotation) dip below, Y_{b+1} comes UP into the up-sampler.
%   Block inputs (sources OUTSIDE/below; the multiply is INSIDE): [U_0;K_0] (global embeddings)
%     and D_b (nested down-sampling) meet at a (x) node INSIDE the block -> D_b[U_0;K_0] -> Pi
%     (and used by Psi); D_b ALSO feeds the GNN Phi; the graph shift S -> Phi.

% ---- vertical levels (single source of truth) -----------------------
\def\Sy{0.8}                        % main chains (block mid-height)
\def\yTop{2.05}                     % I/O top corners: V_b in / Y_b out (raised so the I/O annotations clear the panels)
\def\Ay{1.6}                        % aux top-band level (legacy; panel top now driven by \yPanT)
\def\yPanT{2.75}                    % explicit panel TOP edge anchor (RAISED -> taller top band)
\pgfmathsetmacro{\Sabove}{4.2}      % graph-shift S boxes ABOVE panels; S->top-border gap matched to conditioning->bottom-border gap (~0.83cm)
\def\yHdr{2.75}                     % block headers INSIDE the panels, north-anchored at the top corners (= \yPanT)
\def\yPsi{-1.5}                     % selector Psi_b (stacked under the split; un-squeezed)
\pgfmathsetmacro{\Bmul}{-2.5}       % (x) combine node INSIDE the block (under Pi; riser forks to Psi) -- shifted DOWN
\def\Bsel{-3.45}                    % TopK + junction down/up samplers (LOWERED: more room for Psi/TopK)
\pgfmathsetmacro{\panB}{\Bsel-0.7}  % shared panel bottom edge (just below TopK / the samplers)

\pgfmathsetmacro{\xEin}{\xEpi-\gEio}             % V_b input riser (top-left), OUTSIDE panel
\pgfmathsetmacro{\xEpanL}{\xEpi-1.3}             % encoder panel LEFT-edge anchor (expand panel leftward)
\pgfmathsetmacro{\xEpsi}{\xEsp}                  % selector Psi_b / TopK : under the split

\pgfmathsetmacro{\xEuk}{\xEpi}                   % [U0;K0] directly under Pi/(x) -> vertical into (x)
\pgfmathsetmacro{\xEdb}{\xEgnn+0.6}              % D_b SOURCE under the D_b glyph in Phi^E (post-swap ...;D_b,Theta) -> vertical up
\pgfmathsetmacro{\xEsin}{\xEgnn-0.15}            % S SOURCE (ABOVE panel) over the S glyph in Phi^E -> downward arrow
\pgfmathsetmacro{\xEhdr}{-3.9}                    % encoder header RIGHT edge -> top-right corner (moved right; extends panel right ~0.4cm, clears S arrow + makes room for the C_{b+1} label)
\def\xEdown{-1.2}                                % down-sampling : junction (shifted +0.4 right -> more margin from encoder panel)

\def\xDskip{6.4}                                 % Pi^skip (decoder inner-left)
\pgfmathsetmacro{\xDpi}{\xDskip+3.0}             % decoder projection (wider Pi^skip -> Pi gap)
\pgfmathsetmacro{\xDgnn}{\xDpi+5.2}              % decoder GNN (wider Pi -> Phi^D gap; single-line box)

% ===================== ENCODER (left; signal DESCENDS into the U) =====
\node[hd, anchor=north east] (Ehdr) at (\xEhdr, \yHdr) {Encoder block (depth $b$)};
\node[fuseb] (Epi)  at (\xEpi,  \Sy) {$\Piblk{b}$};
\node[proc]  (Egnn) at (\xEgnn, \Sy) {$\PhiEnc{b}(\bbP_b,\bbS;\bbD_b,\thetaenc_b)$};
\node[splitdot] (Esp) at (\xEsp, \Sy) {};
% selector STACKED under the split: Psi_b (upper) -> TopK (lower)
\node[selb]  (Epsi)  at (\xEpsi, \yPsi) {$\Psisel{b}(\bbZ_b,\bbD_b\UK;\thetasel_{b})$};
\node[topkb] (Etopk) at (\xEpsi, \Bsel) {$\mathrm{TopK}$};

% main input V_b : enters at the TOP-LEFT from OUTSIDE; the bend is far from the box, then a
% long straight run into Pi.west.
\draw[sig] (\xEin,\yTop) |- (Epi.west)
           node[elbl,pos=0.16,above=1pt,align=center] (Vlab) {$\bbV_b{=}\bbC_b\bbZ_{b-1}$\\{\scriptsize(from depth $b{-}1$)}};
% top-row chain (straight)
\draw[sig] (Epi.east)  -- (Egnn.west) node[elbl,pos=0.5,above=2.5pt]{$\bbP_b$};
\draw[sig] (Egnn.east) -- (Esp)       node[elbl,pos=0.55,above=2.5pt]{$\bbZ_b$};
% selector input: Z_b drops STRAIGHT DOWN from the split into Psi.north (no crossing)
\draw[feat] (Esp) -- (Epsi.north) node[elbl,pos=0.55,right=1pt]{$\bbZ_b$};
% scores v_b : Psi_b -> TopK (the un-merge link), straight down
\draw[feat] (Epsi.south) -- (Etopk.north) node[elbl,pos=0.5,right=1pt]{$\bbvv_b$};

% --- conditioning (sources OUTSIDE/below; multiply INSIDE).  [U0;K0] (x) D_b -> D_b[U0;K0] -> Pi
%     (used by Psi); D_b ALSO -> Phi ; S -> Phi. ---
\node[condb] (Euk)  at (\xEuk,  \Bcond)  {$\UK$};
\node[condb] (Edb)  at (\xEdb,  \Bcond)  {$\bbD_b$};
\node[condb] (Esin) at (\xEsin, \Sabove) {$\bbS$};   % S now ABOVE the panel (graph shift fed from the top)
\node[mulb]  (Emul) at (\xEpi,  \Bmul)   {$\times$};
\node[branchdot] (Edbk) at (\xEdb,\Bmul) {};   % D_b branch (to (x) and to Phi)
\node[font=\scriptsize\itshape, text=black!60, anchor=north] at (Euk.south) {global embeddings};
\draw[condarr] (Euk.north) -- (Emul.south);          % [U0;K0] -> (x)  [vertical]
\draw[condarr] (Edb.north) -- (Egnn.south -| Edb);   % D_b -> Phi, vertical into the D_b glyph (post-swap position)
\draw[condarr] (Edbk) -- (Emul.east);                % D_b -> (x)  [horizontal spur off the branch]
\draw[condarr] (Emul.north) -- (Epi.south) node[elbl,pos=0.53]{$\bbD_b\UK$};  % D_b[U0;K0] -> Pi [vertical]
\draw[condarr] (Esin.south) -- (Egnn.north -| Esin);  % S -> Phi from ABOVE, vertical into the S glyph
% D_b[U0;K0] ALSO feeds the selector explicitly: fork the (x)->Pi riser at Psi's level, run right
\node[branchdot] (Eskk) at (\xEpi,\yPsi) {};
\draw[condarr] (Eskk) -- (Epsi.west);

% ---------- down-sampling: OUTSIDE the panel, in the junction ----------
\node[sampb] (Edown) at (\xEdown, \Bsel) {$\downarrow\ \bbC_{b+1}$};

% ===================== DEEP JUNCTION (a small U dipping below) ========

\draw[sig] (\xDup,\Bu) -- (Dup.south) node[elbl,pos=0.45,right=1pt]{$\bbY_{b+1}$};            % UP

% ===================== DECODER (right; FLAT chain, ascends out of the U) ===
\node[hd, anchor=north west] (Dhdr) at (\xDpanL, \yHdr) {Decoder block (depth $b$)};

\draw[condarr] (Dsin.south) -- (Dgnn.north -| Dsin);  % S -> Phi from ABOVE, vertical into the S glyph
